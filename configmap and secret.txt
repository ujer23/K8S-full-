apiVersion: v1
kind: Pod
metadata:
  name: config-demo
spec:
  containers:
    - name: nginx
      image: nginx

      envFrom:
        - configMapRef:
            name: app-config

        - secretRef:
            name: app-secret


apiVersion: v1
kind: Secret
metadata:
  name: app-secret
type: Opaque
stringData:
  USERNAME: "admin"
  PASSWORD: "1234"

apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  APP_NAME: "MyApp"
  APP_ENV: "Production"
